-- 1. T-SQL Program to find factorial of a given number

declare @num int = 5
declare @fact int = 1
declare @i int = 1

while @i <= @num
begin
    set @fact = @fact * @i
    set @i = @i + 1
end

print @fact

-- 2. Create Stored Procedure for Multiplication Table

create or alter proc sp_multiplicationtable(@num int, @limit int)
as
begin
    declare @i int = 1

    while @i <= @limit
    begin
        print @num * @i
        set @i = @i + 1
    end
end
exec sp_multiplicationtable 10,10

-- Create User Defined Function for Student Status

create or alter function fn_studentstatus(@score int)
returns varchar(10)
as
begin
    declare @status varchar(10)

    if @score >= 50
        set @status = 'Pass'
    else
        set @status = 'Fail'

    return @status
end

-- Display Student Details with Status

select 
    s.sid,
    s.sname,
    m.score,
    dbo.fn_studentstatus(m.score) as status
from student s
join marks m
on s.sid = m.sid
