-- Qn-4
select * 
from emptable, depttable
where emptable.edeptno = depttable.deptno
and depttable.dname = 'research'

-- Qn-5
select count(*) as no_of_clerks
from emptable
where ejob = 'clerk'

-- Qn-6
select ejob, avg(esal), count(*)
from emptable
group by ejob

-- Qn-7
select * from emptable
where esal = (select max(esal) from emptable)
or esal = (select min(esal) from emptable)

-- Qn-8
select * from depttable d

on d.deptno = e.edeptno
where e.empno is null

-- Qn-9

and esal > 1200 and edeptno = 20

from emptable, depttable
