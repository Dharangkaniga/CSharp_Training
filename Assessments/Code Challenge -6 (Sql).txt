------ 5 Create a user defined function calculate Bonus for all employees of a  given dept using 	following conditions

    create or alter function fn_DisplayEmployeeSalary(@deptno int,@sal int)
    returns int
    as
    begin
    declare @bonus int;

        --- a.  For Deptno 10 employees 15% of sal as bonus.
        if @deptno=10 
            set @bonus=@sal*15/100
        --- b.  For Deptno 20 employees  20% of sal as bonus
        else if @deptno=20
            set @bonus=@sal*20/100
        --- c   For Others employees 5%of sal as bonus
        else
            set @bonus=@sal*5/100
        return @bonus
    end

    select empno, ename, edeptno, esal,
        dbo.fn_DisplayEmployeeSalary(edeptno, esal)  as bonus
       from emptable;
