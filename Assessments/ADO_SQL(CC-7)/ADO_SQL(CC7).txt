create database Employeesmanagements
use Employeesmanagements

create table EmployeeDetail(
Empno int primary key,
EmpName varchar(30) not null,
EmpSal decimal(10,2) check (EmpSal>=25000),
EmpType char(1) check(EmpType in('F','P'))
)

create or alter procedure InsertionDetail(@EmpName varchar(30),@EmpSal decimal(10,2),@EmpType char(1))
as
begin
	declare @Empno int

	------To Generate empno inside the procedure-----
	select @Empno=isnull(max(Empno),100)+1 from EmployeeDetail

	------To insert some records in Employee_Details-----
	insert into EmployeeDetail(Empno,EmpName,EmpSal,EmpType)
	values(@Empno,@EmpName,@EmpSal,@EmpType)

	print 'Inserted Successfully'
end
